###############    MYSQL   ############### 

CREATE TABLE red (
    id_red INT AUTO_INCREMENT PRIMARY KEY,
    id_recurso_catalogo INT NOT NULL DEFAULT 5, -- Conexión al Catálogo (Red = 5)
    
    -- Configuración basada estrictamente en tu imagen (sin la palabra mes)
    red_tipo ENUM(
        'Normal: Red privada / Hasta 1 Gbps',
        'Plus: Red privada / Hasta 5 Gbps',
        'Premium: Red privada / Hasta 10 Gbps'
    ) NOT NULL,
    
    -- Variable de precio única para este recurso específico
    coste_red DECIMAL(8,2) NOT NULL,
    
    FOREIGN KEY (id_recurso_catalogo) REFERENCES catalogo(id_recurso_catalogo)
);

###############    PHP   ############### 

<?php

class Red 
{
    public int $id_red;
    public int $id_recurso_catalogo;
    public string $red_tipo;
    public float $coste_red;

    // Constructor limpio, sin lógica añadida
    public function __construct(
        int $id_red, 
        int $id_recurso_catalogo, 
        string $red_tipo, 
        float $coste_red
    ) {
        $this->id_red = $id_red;
        $this->id_recurso_catalogo = $id_recurso_catalogo;
        $this->red_tipo = $red_tipo;
        $this->coste_red = $coste_red;
    }

    // Método estático para extraer el registro de la BD y retornar el objeto
    public static function buscarYCrear(PDO $pdo, string $red_tipo): ?Red
    {
        $sql = "SELECT id_red, id_recurso_catalogo, red_tipo, coste_red 
                FROM red 
                WHERE red_tipo = :red_tipo";
                
        $stmt = $pdo->prepare($sql);
        $stmt->execute([':red_tipo' => $red_tipo]);

        $registro = $stmt->fetch(PDO::ATTR_ASSOC);

        if ($registro) {
            return new self(
                (int)$registro['id_red'],
                (int)$registro['id_recurso_catalogo'],
                $registro['red_tipo'],
                (float)$registro['coste_red']
            );
        }

        return null;
    }
}
